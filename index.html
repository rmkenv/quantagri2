Pipeline Architecture

All LLM calls go to https://ollama.com/v1/chat/completions with Authorization: Bearer $OLLAMA_API_KEY. No local GPU. No local daemon. Subscription pricing — free tier available at ollama.com.

curl https://ollama.com/v1/chat/completions \
  -H "Authorization: Bearer $OLLAMA_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "model":   "qwen2.5:7b",
    "messages":[{"role":"user","content":"..."}],
    "stream":  false,
    "format":  "json",
    "options": {"temperature": 0.2}
  }'
import pystac_client, planetary_computer as pc, stackstac

catalog = pystac_client.Client.open(
    "https://planetarycomputer.microsoft.com/api/stac/v1",
    modifier=pc.sign_inplace
)
items = catalog.search(
    collections=["sentinel-2-l2a"],
    bbox=[-96.7, 40.5, -90.1, 43.5],   # Iowa corn
    datetime="2025-04/2025-10",
    query={"eo:cloud_cover": {"lt": 25}}
).item_collection()

stack = stackstac.stack(items, assets=["B04","B08","B11"], resolution=60)
b4  = stack.sel(band="B04") / 10000
b8  = stack.sel(band="B08") / 10000
b11 = stack.sel(band="B11") / 10000
ndvi = (b8 - b4)  / (b8 + b4  + 1e-10)   # vegetation vigor
lswi = (b8 - b11) / (b8 + b11 + 1e-10)   # water stress

ndvi_16d  = ndvi.resample(time="16D").median().mean(dim=["x","y"])
velocity  = ndvi_16d.diff("time")          # dNDVI/dt
on:
